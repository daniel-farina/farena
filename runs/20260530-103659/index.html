(function () {
  const currentEl = document.getElementById('current');
  const historyEl = document.getElementById('history');

  let current = '0';      // string being entered / shown
  let previous = null;    // stored operand (number)
  let operator = null;    // pending operator
  let resetNext = false;  // next digit starts a fresh number

  const opSymbols = { '+': '+', '-': '−', '*': '×', '/': '÷' };

  function format(n) {
    if (n === 'Error') return n;
    const num = Number(n);
    if (!isFinite(num)) return 'Error';
    // Limit precision to avoid floating-point noise, keep it readable.
    let str = Math.abs(num) < 1e-12 ? '0'
      : parseFloat(num.toPrecision(12)).toString();
    return str;
  }

  function render() {
    currentEl.textContent = current.length > 0 ? current : '0';
    if (operator && previous !== null) {
      historyEl.textContent = `${format(previous)} ${opSymbols[operator]}`;
    } else {
      historyEl.textContent = '';
    }
    highlightOp();
  }

  function highlightOp() {
    document.querySelectorAll('.op').forEach(b => b.classList.remove('active'));
    if (operator && !resetNext === false) {
      const btn = document.querySelector(`.op[data-op="${operator}"]`);
      if (btn) btn.classList.add('active');
    }
  }

  function inputNumber(d) {
    if (resetNext) { current = d; resetNext = false; return; }
    if (current === '0') current = d;
    else current += d;
  }

  function inputDecimal() {
    if (resetNext) { current = '0.'; resetNext = false; return; }
    if (!current.includes('.')) current += '.';
  }

  function compute(a, b, op) {
    switch (op) {
      case '+': return a + b;
      case '-': return a - b;
      case '*': return a * b;
      case '/': return b === 0 ? 'Error' : a / b;
    }
  }

  function handleOperator(nextOp) {
    const inputVal = parseFloat(current);
    if (operator && !resetNext) {
      const result = compute(previous, inputVal, operator);
      if (result === 'Error') { showError(); return; }
      previous = parseFloat(format(result));
      current = format(result);
    } else {
      previous = inputVal;
    }
    operator = nextOp;
    resetNext = true;
  }

  function handleEquals() {
    if (operator === null || previous === null) return;
    const inputVal = parseFloat(current);
    const result = compute(previous, inputVal, operator);
    if (result === 'Error') { showError(); return; }
    historyEl.textContent = `${format(previous)} ${opSymbols[operator]} ${format(inputVal)} =`;
    current = format(result);
    previous = null;
    operator = null;
    resetNext = true;
    currentEl.textContent = current;
    highlightOp();
    return true;
  }

  function showError() {
    current = 'Error';
    previous = null;
    operator = null;
    resetNext = true;
    render();
    current = '0';
  }

  function clearAll() {
    current = '0';
    previous = null;
    operator = null;
    resetNext = false;
  }

  function toggleSign() {
    if (current === '0' || current === 'Error') return;
    current = current.startsWith('-') ? current.slice(1) : '-' + current;
  }

  function percent() {
    current = format(parseFloat(current) / 100);
    resetNext = true;
  }

  // Click handling
  document.querySelector('.keys').addEventListener('click', (e) => {
    const t = e.target.closest('button');
    if (!t) return;
    const acClear = document.querySelector('[data-action="clear"]');

    if (t.dataset.num !== undefined) { inputNumber(t.dataset.num); acClear.textContent = 'C'; }
    else if (t.dataset.op) { handleOperator(t.dataset.op); }
    else {
      switch (t.dataset.action) {
        case 'equals': if (handleEquals()) { acClear.textContent = 'AC'; return; } break;
        case 'clear': clearAll(); acClear.textContent = 'AC'; break;
        case 'sign': toggleSign(); break;
        case 'percent': percent(); break;
        case 'decimal': inputDecimal(); acClear.textContent = 'C'; break;
      }
    }
    render();
  });

  // Keyboard support
  window.addEventListener('keydown', (e) => {
    const k = e.key;
    if (k >= '0' && k <= '9') { inputNumber(k); }
    else if (k === '.') { inputDecimal(); }
    else if (['+', '-', '*', '/'].includes(k)) { handleOperator(k); }
    else if (k === 'Enter' || k === '=') { e.preventDefault(); handleEquals(); }
    else if (k === 'Escape') { clearAll(); }
    else if (k === 'Backspace') {
      if (!resetNext && current !== '0' && current !== 'Error') {
        current = current.length > 1 ? current.slice(0, -1) : '0';
      }
    }
    else if (k === '%') { percent(); }
    else return;
    render();
  });

  render();
})();

    function evaluateExpression(expr) {
      if (!expr || expr.trim() === '') return 0;

      // Replace display symbols with JS operators
      let sanitized = expr
        .replace(/×/g, '*')
        .replace(/÷/g, '/')
        .replace(/−/g, '-')
        .replace(/π/g, 'Math.PI')
        .replace(/e(?![a-zA-Z])/g, 'Math.E');

      // Handle implicit multiplication (e.g., 2(3+1) or (2+3)(4+1))
      sanitized = sanitized.replace(/(\d)\(/g, '$1*(');
      sanitized = sanitized.replace(/\)\(/g, ')*(');
      sanitized = sanitized.replace(/(\d)([a-zA-Z(])/g, '$1*$2');

      const validPattern = /^[0-9+\-*/.() \tMath.PIe]+$/;

        }

        throw new Error('Syntax error');
      }

      if (state.display.length > 10) {
        resultEl.classList.add('small');
      } else {
        resultEl.classList.remove('small');
      }

      // Error styling
      if (state.hasError) {
        resultEl.classList.add('error');
      } else {
        resultEl.classList.remove('error');
      }

      // Memory indicator
      if (Math.abs(state.memory) > 1e-10) {
        memoryIndicator.classList.add('active');
      } else {
        memoryIndicator.classList.remove('active');
      }
    }

    function setDisplay(value) {
      if (typeof value === 'number') {
        // Smart formatting
        if (Math.abs(value) > 1e12 || (Math.abs(value) < 1e-6 && value !== 0)) {
          state.display = value.toExponential(6);
        } else {
          // Remove unnecessary trailing zeros
          let str = value.toPrecision(12);
          str = parseFloat(str).toString();
          if (str.length > 14) str = value.toExponential(6);
          state.display = str;
        }
      } else {
        state.display = String(value);
      }
    }

    // ==================== CORE OPERATIONS ====================
    function inputDigit(digit) {
      if (state.hasError) clearAll();

      if (state.waitingForOperand) {
        state.display = String(digit);
        state.waitingForOperand = false;
      } else {
        state.display = state.display === '0' ? String(digit) : state.display + digit;
      }
      updateDisplay();
    }

    function inputDecimal() {
      if (state.hasError) clearAll();

      if (state.waitingForOperand) {
        state.display = '0.';
        state.waitingForOperand = false;
      } else if (!state.display.includes('.')) {
        state.display += '.';
      }
      updateDisplay();
    }

    function clearAll() {
      state.display = '0';
      state.expression = '';
      state.previousValue = null;
      state.operator = null;
      state.waitingForOperand = false;
      state.hasError = false;
      updateDisplay();
    }

    function clearEntry() {
      if (state.hasError) {
        clearAll();
        return;
      }
      state.display = '0';
      updateDisplay();
    }

    function backspace() {
      if (state.hasError || state.waitingForOperand) return;

      if (state.display.length === 1 || (state.display.length === 2 && state.display.startsWith('-'))) {
        state.display = '0';
      } else {
        state.display = state.display.slice(0, -1);
      }
      updateDisplay();
    }

    function toggleSign() {
      if (state.hasError || state.display === '0') return;

      if (state.display.startsWith('-')) {
        state.display = state.display.slice(1);
      } else {
        state.display = '-' + state.display;
      }
      updateDisplay();
    }

    function performCalculation() {
      if (state.operator === null || state.previousValue === null || state.waitingForOperand) {
        return;
      }

      const inputValue = parseFloat(state.display);
      let result;

      try {
        switch (state.operator) {
          case '+': result = state.previousValue + inputValue; break;
          case '−': result = state.previousValue - inputValue; break;
          case '×': result = state.previousValue * inputValue; break;
          case '÷':
            if (inputValue === 0) throw new Error('Cannot divide by zero');
            result = state.previousValue / inputValue;
            break;
          case '^': result = Math.pow(state.previousValue, inputValue); break;
          case 'mod': result = state.previousValue % inputValue; break;
          default: return;
        }

        if (!isFinite(result)) throw new Error('Math error');

        // Add to history
        const expr = `${formatNumber(state.previousValue)} ${state.operator} ${formatNumber(inputValue)}`;
        addToHistory(expr, result);

        state.display = String(result);
        state.previousValue = result;
        state.lastResult = result;
        state.expression = '';
        state.operator = null;
        state.waitingForOperand = true;
        state.hasError = false;

      } catch (e) {
        state.display = e.message;
        state.hasError = true;
        state.expression = '';
        state.operator = null;
        state.previousValue = null;
      }

      updateDisplay();
    }

    function setOperator(nextOperator) {
      if (state.hasError) return;

      const inputValue = parseFloat(state.display);

      if (state.previousValue === null) {
        state.previousValue = inputValue;
      } else if (state.operator && !state.waitingForOperand) {
        performCalculation();
        state.previousValue = parseFloat(state.display);
      }

      state.operator = nextOperator;
      state.waitingForOperand = true;

      const opSymbol = nextOperator === '^' ? '^' : nextOperator;
      state.expression = `${formatNumber(state.previousValue)} ${opSymbol}`;
      updateDisplay();
    }

    function formatNumber(num) {
      if (typeof num !== 'number') return String(num);
      return Number.isInteger(num) ? String(num) : num.toPrecision(8).replace(/\.?0+$/, '');
    }

    // ==================== SCIENTIFIC FUNCTIONS ====================
    function applyFunction(fnName) {
      if (state.hasError) return;

      let value = parseFloat(state.display);
      let result;
      let displayExpr = '';

      try {
        switch (fnName) {
          case 'sin': result = scientificFns.sin(value); displayExpr = `sin(${value})`; break;
          case 'cos': result = scientificFns.cos(value); displayExpr = `cos(${value})`; break;
          case 'tan': result = scientificFns.tan(value); displayExpr = `tan(${value})`; break;
          case 'asin': result = scientificFns.asin(value); displayExpr = `asin(${value})`; break;
          case 'acos': result = scientificFns.acos(value); displayExpr = `acos(${value})`; break;
          case 'atan': result = scientificFns.atan(value); displayExpr = `atan(${value})`; break;
          case 'log': 
            if (value <= 0) throw new Error('Domain error');
            result = scientificFns.log(value); displayExpr = `log(${value})`; 
            break;
          case 'ln': 
            if (value <= 0) throw new Error('Domain error');
            result = scientificFns.ln(value); displayExpr = `ln(${value})`; 
            break;
          case 'sqrt':
            if (value < 0) throw new Error('Domain error');
            result = scientificFns.sqrt(value); displayExpr = `√(${value})`; 
            break;
          case 'square': result = scientificFns.square(value); displayExpr = `(${value})²`; break;
          case 'cube': result = scientificFns.cube(value); displayExpr = `(${value})³`; break;
          case 'reciprocal':
            if (value === 0) throw new Error('Divide by zero');
            result = scientificFns.reciprocal(value); displayExpr = `1/(${value})`; 
            break;
          case 'percent': result = scientificFns.percent(value); displayExpr = `${value}%`; break;
          case 'factorial':
            if (value < 0 || value > 170) throw new Error('Domain error');
            result = scientificFns.factorial(Math.floor(value)); displayExpr = `${Math.floor(value)}!`; 
            break;
          case 'negate': result = -value; displayExpr = `-(${value})`; break;
          default: return;
        }

        if (!isFinite(result)) throw new Error('Math error');

        addToHistory(displayExpr, result);

        state.display = String(result);
        state.lastResult = result;
        state.expression = displayExpr;
        state.waitingForOperand = true;

      } catch (e) {
        state.display = e.message;
        state.hasError = true;
        state.expression = '';
      }

      updateDisplay();
    }

    function applyConstant(name) {
      if (state.hasError) clearAll();

      let val;
      if (name === 'pi') val = Math.PI;
      else if (name === 'e') val = Math.E;

      if (state.waitingForOperand || state.display === '0') {
        state.display = String(val);
      } else {
        // Implicit multiply if typing before constant
        const current = parseFloat(state.display);
        state.display = String(current * val);
      }
      state.waitingForOperand = false;
      updateDisplay();
    }

    // ==================== HISTORY ====================
    function addToHistory(expression, result) {
      const entry = {
        expr: expression,
        result: typeof result === 'number' ? Number(result.toPrecision(12)) : result,
        timestamp: Date.now()
      };
      
      state.history.unshift(entry);
      
      // Keep last 20
      if (state.history.length > 20) state.history.pop();

      renderHistory();
    }

    function renderHistory() {
      if (state.history.length === 0) {
        historyList.innerHTML = `<div class="empty-history">No history yet</div>`;
        return;
      }

      historyList.innerHTML = state.history.map((entry, idx) => {
        const resStr = typeof entry.result === 'number' 
          ? (Math.abs(entry.result) > 1e9 ? entry.result.toExponential(4) : entry.result.toString())
          : entry.result;
        return `
          <div class="history-item" data-index="${idx}">
            <span class="history-expr">${entry.expr}</span>
            <span class="history-result">${resStr}</span>
          </div>
        `;
      }).join('');

      // Attach click handlers
      historyList.querySelectorAll('.history-item').forEach(item => {
        item.addEventListener('click', () => {
          const idx = parseInt(item.dataset.index);
          const entry = state.history[idx];
          if (!entry) return;

          if (state.hasError) clearAll();

          // Load result into display
          state.display = String(entry.result);
          state.lastResult = entry.result;
          state.waitingForOperand = true;
          state.expression = entry.expr;
          updateDisplay();
        });
      });
    }

    function clearHistory() {
      state.history = [];
      renderHistory();
    }

    // ==================== EVALUATE FULL EXPRESSION ====================
    function evaluateFullExpression() {
      if (!state.expression || state.hasError) return;

      // If we have a pending operator + display value, complete it
      if (state.operator && !state.waitingForOperand) {
        performCalculation();
        return;
      }

      // Otherwise try to evaluate the current expression display
      try {
        let toEval = state.expression || state.display;
        
        // If expression is empty but we have a display, try display
        if (!toEval || toEval === '0') toEval = state.display;

        const result = evaluateExpression(toEval);

        addToHistory(toEval, result);

        state.display = String(result);
        state.lastResult = result;
        state.expression = '';
        state.operator = null;
        state.previousValue = null;
        state.waitingForOperand = true;
        state.hasError = false;

      } catch (e) {
        state.display = 'Syntax error';
        state.hasError = true;
        state.expression = '';
      }

      updateDisplay();
    }

    function handleKeyDown(e) {
      if (state.hasError && !['Escape', 'c', 'C', 'Backspace'].includes(e.key)) {
        clearAll();
      }

      switch (e.key) {
        case '0': case '1': case '2': case '3': case '4':
        case '5': case '6': case '7': case '8': case '9':
          inputDigit(parseInt(e.key));
          break;
        case '.':
          inputDecimal();
          break;
        case '+':
          setOperator('+');
          break;
        case '-':
          setOperator('−');
          break;
        case '*':
          setOperator('×');
          break;
        case '/':
          e.preventDefault();
          setOperator('÷');
          break;
        case '^':
          power();
          break;
        case '%':
          applyFunction('percent');
          break;
        case 'Enter':
        case '=':
          e.preventDefault();
          evaluateFullExpression();
          break;
        case 'Escape':
        case 'c':
        case 'C':
          clearAll();
          break;
        case 'Backspace':
          backspace();
          break;
        case 'Delete':
          clearEntry();
          break;
        case 's':
        case 'S':
          if (state.scientific) applyFunction('sin');
          break;
        case 'o':
        case 'O':
          if (state.scientific) applyFunction('cos');
          break;
        case 't':
        case 'T':
          if (state.scientific) applyFunction('tan');
          break;
        case 'r':
        case 'R':
          memoryRecall();
          break;
        case 'm':
        case 'M':
          if (e.shiftKey) memorySubtract();
          else memoryAdd();
          break;
        case 'p':
        case 'P':
          if (state.scientific) applyConstant('pi');
          break;
      }
    }

    // ==================== PASTE SUPPORT ====================
    async function handlePaste(e) {
      e.preventDefault();
      const text = (e.clipboardData || window.clipboardData).getData('text').trim();
      
      if (!text) return;

      // Try to evaluate pasted expression
      try {
        const result = evaluateExpression(text);
        addToHistory(text, result);
        state.display = String(result);
        state.lastResult = result;
        state.waitingForOperand = true;
        state.expression = text;
        state.hasError = false;
      } catch {
        // If not a valid expression, try to parse as number
        const num = parseFloat(text);
        if (!isNaN(num)) {
          state.display = String(num);
          state.waitingForOperand = false;
        }
      }
      updateDisplay();
    }

    function performCalculation() {
      if (state.operator === null || state.previousValue === null || state.waitingForOperand) {
        return;
      }

      const inputValue = parseFloat(state.display);
      let result;

      try {
        switch (state.operator) {
          case '+': result = state.previousValue + inputValue; break;
          case '−': result = state.previousValue - inputValue; break;
          case '×': result = state.previousValue * inputValue; break;
          case '÷':
            if (inputValue === 0) throw new Error('Cannot divide by zero');
            result = state.previousValue / inputValue;
            break;
          case '^': result = Math.pow(state.previousValue, inputValue); break;
          case 'mod': result = state.previousValue % inputValue; break;
          default: return;
        }

        if (!isFinite(result)) throw new Error('Math error');

        // Add to history
        const expr = `${formatNumber(state.previousValue)} ${state.operator} ${formatNumber(inputValue)}`;
        addToHistory(expr, result);

        state.display = String(result);
        state.previousValue = result;
        state.lastResult = result;
        state.expression = '';
        state.operator = null;
        state.waitingForOperand = true;
        state.hasError = false;

      } catch (e) {
        state.display = e.message;
        state.hasError = true;
        state.expression = '';
        state.operator = null;
        state.previousValue = null;
      }

      updateDisplay();
    }

    function setOperator(nextOperator) {
      if (state.hasError) return;

      const inputValue = parseFloat(state.display);

      if (state.previousValue === null) {
        state.previousValue = inputValue;
      } else if (state.operator && !state.waitingForOperand) {
        performCalculation();
        state.previousValue = parseFloat(state.display);
      }

      state.operator = nextOperator;
      state.waitingForOperand = true;

      const opSymbol = nextOperator === '^' ? '^' : nextOperator;
      state.expression = `${formatNumber(state.previousValue)} ${opSymbol}`;
      updateDisplay();
    }

    function formatNumber(num) {
      if (typeof num !== 'number') return String(num);
      return Number.isInteger(num) ? String(num) : num.toPrecision(8).replace(/\.?0+$/, '');
    }

          case 'acos': result = scientificFns.acos(value); displayExpr = `acos(${value})`; break;
          case 'atan': result = scientificFns.atan(value); displayExpr = `atan(${value})`; break;
          case 'log': 
            if (value <= 0) throw new Error('Domain error');
            result = scientificFns.log(value); displayExpr = `log(${value})`; 
            break;
          case 'ln': 
            if (value <= 0) throw new Error('Domain error');
            result = scientificFns.ln(value); displayExpr = `ln(${value})`;

            if (value < 0) throw new Error('Domain error');
            result = scientificFns.sqrt(value); displayExpr = `√(${value})`;

          case 'square': result = scientificFns.square(value); displayExpr = `(${value})²`; break;
          case 'cube': result = scientificFns.cube(value); displayExpr = `(${value})³`; break;
          case 'reciprocal':
            if (value === 0) throw new Error('Divide by zero');
            result = scientificFns.reciprocal(value); displayExpr = `1/(${value})`; 
            break;
          case 'percent': result = scientificFns.percent(value); displayExpr = `${value}%`; break;
          case 'factorial':
            if (value < 0 || value > 170) throw new Error('Domain error');

        }

        addToHistory(displayExpr, result);

        state.display = String(result);
        state.lastResult = result;
        state.expression = displayExpr;
        state.waitingForOperand = true;

        state.hasError = true;
        state.expression = '';

    function evaluateFullExpression() {
      if (!state.expression || state.hasError) return;

      // If we have a pending operator + display value, complete it
      if (state.operator && !state.waitingForOperand) {
        performCalculation();
        return;
      }

      // Otherwise try to evaluate the current expression display
      try {
        let toEval = state.expression || state.display;
        
        // If expression is empty but we have a display, try display
        if (!toEval || toEval === '0') toEval = state.display;

        const result = evaluateExpression(toEval);

        addToHistory(toEval, result);

        state.display = String(result);
        state.lastResult = result;
        state.expression = '';
        state.operator = null;
        state.previousValue = null;
        state.waitingForOperand = true;
        state.hasError = false;

      } catch (e) {
        state.display = 'Syntax error';
        state.hasError = true;
        state.expression = '';
      }

      updateDisplay();
    }
