(function () {
  const currentEl = document.getElementById('current');
  const historyEl = document.getElementById('history');

  let current = '0';      // string being typed
  let previous = null;    // stored operand (number)
  let operator = null;    // pending operator
  let justEvaluated = false;

  const ops = {
    '+': (a, b) => a + b,
    '-': (a, b) => a - b,
    '*': (a, b) => a * b,
    '/': (a, b) => b === 0 ? NaN : a / b,
  };
  const symbols = { '+': '+', '-': '−', '*': '×', '/': '÷' };

  function format(n) {
    if (!isFinite(n)) return 'Error';
    // Trim float noise, keep reasonable precision.
    const rounded = Math.round((n + Number.EPSILON) * 1e10) / 1e10;
    return String(rounded);
  }

  function render() {
    currentEl.textContent = current;
    historyEl.textContent =
      previous !== null && operator
        ? `${format(previous)} ${symbols[operator]}`
        : '';
  }

  function inputDigit(d) {
    if (justEvaluated) { current = '0'; justEvaluated = false; }
    current = current === '0' ? d : current + d;
  }

  function inputDecimal() {
    if (justEvaluated) { current = '0'; justEvaluated = false; }
    if (!current.includes('.')) current += '.';
  }

  function chooseOperator(nextOp) {
    const value = parseFloat(current);
    if (previous === null) {
      previous = value;
    } else if (operator && !justEvaluated) {
      previous = ops[operator](previous, value);
    }
    operator = nextOp;
    current = format(previous);
    justEvaluated = false;
    // After choosing an operator, next digit starts fresh.
    pendingFresh = true;
  }

  let pendingFresh = false;

  function inputDigitGuarded(d) {
    if (pendingFresh) { current = '0'; pendingFresh = false; }
    inputDigit(d);
  }
  function inputDecimalGuarded() {
    if (pendingFresh) { current = '0'; pendingFresh = false; }
    inputDecimal();
  }

  function equals() {
    if (operator === null || previous === null) return;
    const value = parseFloat(current);
    const result = ops[operator](previous, value);
    historyEl.textContent = `${format(previous)} ${symbols[operator]} ${format(value)} =`;
    current = format(result);
    previous = null;
    operator = null;
    justEvaluated = true;
    pendingFresh = false;
    currentEl.textContent = current;
  }

  function clearAll() {
    current = '0';
    previous = null;
    operator = null;
    justEvaluated = false;
    pendingFresh = false;
  }

  function toggleSign() {
    if (current === '0') return;
    current = current.startsWith('-') ? current.slice(1) : '-' + current;
  }

  function percent() {
    current = format(parseFloat(current) / 100);
    justEvaluated = true;
  }

  document.querySelector('.keys').addEventListener('click', (e) => {
    const btn = e.target.closest('button');
    if (!btn) return;

    if (btn.dataset.num !== undefined) {
      inputDigitGuarded(btn.dataset.num);
    } else if (btn.dataset.op !== undefined) {
      chooseOperator(btn.dataset.op);
    } else {
      switch (btn.dataset.action) {
        case 'decimal': inputDecimalGuarded(); break;
        case 'equals': equals(); return;
        case 'clear': clearAll(); break;
        case 'sign': toggleSign(); break;
        case 'percent': percent(); break;
      }
    }
    render();
  });

  // Keyboard support
  window.addEventListener('keydown', (e) => {
    if (e.key >= '0' && e.key <= '9') inputDigitGuarded(e.key);
    else if (e.key === '.') inputDecimalGuarded();
    else if (['+', '-', '*', '/'].includes(e.key)) chooseOperator(e.key);
    else if (e.key === 'Enter' || e.key === '=') { e.preventDefault(); equals(); return; }
    else if (e.key === 'Escape') clearAll();
    else if (e.key === '%') percent();
    else if (e.key === 'Backspace') {
      current = current.length > 1 ? current.slice(0, -1) : '0';
    } else return;
    render();
  });

  render();
})();
